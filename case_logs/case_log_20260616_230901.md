### **Oracle Unified Auditing (OUA) - Target DB (172.18.18.2)**

```sql

    EVENT_TIMESTAMP,
    DBUSER_NAME,
    CLIENT_IDENTIFIER,
    ACTION_NAME,
    RETURN_CODE,
    SQL_TEXT
FROM UNIFIED_AUDIT_TRAIL

    CLIENT_IP = '10.0.0.4'
    AND EVENT_TIMESTAMP >= TRUNC(SYSDATE) - 1
    AND (RETURN_CODE != 0 OR ACTION_NAME IN ('LOGON', 'CREATE USER', 'GRANT'))

-- 2. Identify privilege escalation or role changes
SELECT 
    EVENT_TIMESTAMP,
    DBUSER_NAME,
    ACTION_NAME,
    SQL_TEXT
FROM UNIFIED_AUDIT_TRAIL
WHERE 
    ACTION_NAME IN ('GRANT', 'CREATE USER', 'ALTER USER', 'SET ROLE')
    AND EVENT_TIMESTAMP >= TRUNC(SYSDATE) - 1
ORDER BY EVENT_TIMESTAMP DESC;

-- 3. Detect data exfiltration (SELECT on sensitive tables)
SELECT 
    EVENT_TIMESTAMP,
    DBUSER_NAME,
    CLIENT_IP,
    SQL_TEXT,
    RETURN_CODE
FROM UNIFIED_AUDIT_TRAIL
WHERE 
    ACTION_NAME = 'SELECT'
    AND OBJECT_NAME IN ('CUSTOMERS', 'FINANCIAL_DATA', 'PII_TABLE')
    AND EVENT_TIMESTAMP >= TRUNC(SYSDATE) - 1
ORDER BY EVENT_TIMESTAMP DESC;

-- 4. Check for failed authentication attempts
SELECT 
    EVENT_TIMESTAMP,
    DBUSER_NAME,
    CLIENT_IP,
    RETURN_CODE,
    ACTION_NAME
FROM UNIFIED_AUDIT_TRAIL
WHERE 
    RETURN_CODE IN (1017, 28000)  -- Invalid credentials / Account locked
    AND EVENT_TIMESTAMP >= TRUNC(SYSDATE) - 1
ORDER BY EVENT_TIMESTAMP DESC;

-- 5. Identify unusual connection patterns
SELECT 
    CLIENT_IP,
    COUNT(*) as connection_count,
    MIN(EVENT_TIMESTAMP) as first_seen,
    MAX(EVENT_TIMESTAMP) as last_seen
FROM UNIFIED_AUDIT_TRAIL
WHERE 
    ACTION_NAME = 'LOGON'
    AND EVENT_TIMESTAMP >= TRUNC(SYSDATE) - 1
GROUP BY CLIENT_IP
HAVING COUNT(*) > 50
ORDER BY connection_count DESC;
```
